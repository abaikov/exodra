```ts
import { EXO_PORT_EVENT, EXO_PORT_TEXT, createExoMachineBuilder, createExoSchema, exoPorts } from '@exodra/core';
import type { TExoMachineNode } from '@exodra/core';
import { ExoDomBackend, withExoCompiler } from '@exodra/dom';

const CounterPorts = exoPorts({ count: EXO_PORT_TEXT, increment: EXO_PORT_EVENT });

const counterSchema = createExoSchema(CounterPorts, (b) => {
    b.el('div', { class: 'counter' }, (div) => {
        div.el('span', null, (span) => {
            span.textPort(CounterPorts.count);
        });
        div.el('button', { type: 'button' }, (button) => {
            button.eventPort(CounterPorts.increment, 'click');
            button.text('+1');
        });
    });
});

const counterBuilder = createExoMachineBuilder<TExoMachineNode, unknown, { count: number }>(counterSchema);
counterBuilder.createInitialState(() => ({ count: 0 }));
counterBuilder.wire((c) => {
    c.text(CounterPorts.count, String(c.state.count));
});
counterBuilder.on(CounterPorts.increment, (c) => {
    c.state.count += 1;
    c.text(CounterPorts.count, String(c.state.count));
});

withExoCompiler(new ExoDomBackend()).render(counterBuilder.done(), document.body);
```
