```ts
import { EXO_PORT_EVENT, EXO_PORT_TEXT, composeExoMachine, createExoMachine, createExoSchema, exoPorts, withExoHandlers, withExoTransitions, withExoWire } from '@exodra/core';
import { ExoDomBackend, withExoCompiler } from '@exodra/dom';

const CounterPorts = exoPorts({ COUNT: EXO_PORT_TEXT, INCREMENT: EXO_PORT_EVENT });

const CounterSchema = createExoSchema(CounterPorts, (b) => {
    b.el('div', { class: 'counter' }, (div) => {
        div.el('span', null, (span) => span.textPort(CounterPorts.COUNT));
        div.el('button', { type: 'button' }, (button) => {
            button.eventPort(CounterPorts.INCREMENT, 'click');
            button.text('+1');
        });
    });
});

export const counterMachine = composeExoMachine(
    createExoMachine(CounterSchema, { createInitialState: () => ({ count: 0 }) }),
    (m) => withExoTransitions(m, {
        increment: (c) => {
            c.state.count += 1;
            c.text(CounterPorts.COUNT, String(c.state.count));
        },
    }),
    (m) => withExoWire(m, (c) => c.text(CounterPorts.COUNT, String(c.state.count))),
    (m) => withExoHandlers(m, (on) => on(CounterPorts.INCREMENT, (c) => increment(c))),
);
export const { increment } = counterMachine.transitions;

withExoCompiler(new ExoDomBackend()).render(counterMachine, document.body);
```
