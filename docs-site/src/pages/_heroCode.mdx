```tsx
import { exoJsx } from '@exodra/jsx';
import {
    composeExoMachine, createExoMachine, withExoHandlers, withExoTransitions, withExoWire,
} from '@exodra/core';
import type { TExoEventPort, TExoTextPort } from '@exodra/core';
import { ExoDomBackend, withExoBuiltSchemas } from '@exodra/dom';

const CounterSchema = exoJsx<{ COUNT: TExoTextPort; INCREMENT: TExoEventPort }>(($) => (
    <div class="counter">
        <span>{$.COUNT}</span>
        <button type="button" onClick={$.INCREMENT}>+1</button>
    </div>
));
const CounterPorts = CounterSchema.ports;

export const counterMachine = composeExoMachine(
    createExoMachine(CounterSchema, { createInitialState: () => ({ count: 0 }) }),
    (m) => withExoTransitions(m, {
        increment: (c) => {
            c.state.count += 1;
            c.text(CounterPorts.COUNT, String(c.state.count));
        },
    }),
    (m) => withExoWire(m, (c) => c.text(CounterPorts.COUNT, String(c.state.count))),
    (m) => {
        const { increment } = m.transitions;
        return withExoHandlers(m, (on) => on(CounterPorts.INCREMENT, (c) => increment(c)));
    },
);
export const { increment } = counterMachine.transitions;

withExoBuiltSchemas(new ExoDomBackend()).render(counterMachine, document.body);
```
